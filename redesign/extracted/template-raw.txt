const { useState } = React;

function App() {
  const [screen, setScreen] = useState('login');
  const [selectedWs, setSelectedWs] = useState(null);

  const navigate = (to, data = {}) => {
    setScreen(to);
    if (data.ws !== undefined) setSelectedWs(data.ws);
  };

  const ctx = { screen, navigate, selectedWs };

  const screens = {
    login:   <ProtoLoginScreen />,
    home:    <ProtoHomeScreen />,
    create:  <ProtoCreateScreen />,
    info:    <ProtoInfoScreen />,
    report:  <ProtoReportScreen />,
    diary:   <ProtoDiaryScreen />,
    setting: <ProtoSettingScreen />,
  };

  return (
    <NavContext.Provider value={ctx}>
      {screens[screen] || screens['home']}
    </NavContext.Provider>
  );
}

ReactDOM.createRoot(document.getElementById('root')).render(<App />);
